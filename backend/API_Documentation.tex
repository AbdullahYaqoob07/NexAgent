% Introduction
\chapter{Introduction}

\section{Overview}

NexAgent is a comprehensive workflow automation platform providing enterprise-grade APIs for managing workflows, templates, integrations, marketplace, notifications, analytics, audit logs, and billing operations. This document provides detailed technical documentation for all API modules.

\section{Base URL}

\textbf{Production:} \texttt{https://api.nexagent.com}\\
\textbf{Development:} \texttt{http://localhost:8000}

\section{API Version}

Current API Version: \texttt{v1}\\
All endpoints are prefixed with: \texttt{/api/v1}

\section{Authentication}

NexAgent uses Firebase Authentication with Bearer token authorization.

\begin{itemize}[leftmargin=*]
    \item \textbf{Method:} Bearer Token Authentication
    \item \textbf{Header:} \texttt{Authorization: Bearer <token>}
    \item \textbf{Token Type:} Firebase ID Token
    \item \textbf{Session Duration:} 1 week (configurable)
\end{itemize}

\begin{infobox}
\textbf{Security Measures:}
\begin{itemize}[leftmargin=*]
    \item Single session enforcement (one device at a time)
    \item Automatic token expiration after 7 days
    \item Session invalidation on logout
    \item Trusted host middleware enabled
    \item CORS protection configured
\end{itemize}
\end{infobox}

% Authentication Module
\chapter{Authentication Module}

\section{Module Overview}

The Authentication module handles user registration, login, password recovery, token verification, and session management using Firebase Authentication.

\subsection{Base Path}
\texttt{/api/v1/auth}

\subsection{Module Tag}
\texttt{Authentication}

\section{Endpoints}

\subsection{Sign Up}

\textbf{POST} \texttt{/api/v1/auth/signup}

\subsubsection{Description}
Create a new user account with email verification.

\subsubsection{Use Case}
User registration for new platform members. Upon successful registration, users receive a verification email and can access the platform with limited privileges until email is verified.

\subsubsection{Request Body}
\begin{lstlisting}[style=jsoncode,language=json]
{
  "email": "user@example.com",
  "password": "SecurePass123!",
  "display_name": "John Doe"
}
\end{lstlisting}

\subsubsection{Response}
\begin{lstlisting}[style=jsoncode,language=json]
{
  "success": true,
  "message": "Account created successfully. Please verify your email.",
  "user": {
    "uid": "firebase_user_id",
    "email": "user@example.com",
    "display_name": "John Doe",
    "email_verified": false
  }
}
\end{lstlisting}

\begin{infobox}
\textbf{Important Information:}
\begin{itemize}[leftmargin=*]
    \item Minimum password length: 8 characters
    \item Password must contain uppercase, lowercase, and numbers
    \item Email verification required for full access
    \item Rate limit: 100 requests per minute
    \item HTTP Status: 201 Created on success
\end{itemize}
\end{infobox}

\subsection{Sign In}

\textbf{POST} \texttt{/api/v1/auth/signin}

\subsubsection{Description}
Authenticate user with email and password, creating a session token.

\subsubsection{Use Case}
User login with session creation. Single session enforcement ensures only one active device per user. Previous sessions are automatically invalidated.

\subsubsection{Request Body}
\begin{lstlisting}[style=jsoncode,language=json]
{
  "email": "user@example.com",
  "password": "SecurePass123!"
}
\end{lstlisting}

\subsubsection{Response}
\begin{lstlisting}[style=jsoncode,language=json]
{
  "success": true,
  "message": "Sign in successful. Session created (1 week validity).",
  "user": {
    "uid": "firebase_user_id",
    "email": "user@example.com",
    "display_name": "John Doe",
    "email_verified": true
  },
  "access_token": "session_token_string"
}
\end{lstlisting}

\begin{infobox}
\textbf{Session Information:}
\begin{itemize}[leftmargin=*]
    \item Session validity: 7 days
    \item Single active session per user
    \item Device information tracked
    \item IP address logged for security
    \item Auto-invalidates previous sessions
\end{itemize}
\end{infobox}

\subsection{Verify Token}

\textbf{POST} \texttt{/api/v1/auth/verify-token}

\subsubsection{Description}
Verify Firebase ID token and create/refresh session.

\subsubsection{Use Case}
Used by frontend applications to verify Firebase authentication state and obtain NexAgent session token. Essential for maintaining authenticated state across page reloads.

\subsection{Forgot Password}

\textbf{POST} \texttt{/api/v1/auth/forgot-password}

\subsubsection{Description}
Send password reset email to registered user.

\subsubsection{Use Case}
Password recovery flow. Sends secure password reset link via email without revealing account existence for security purposes.

\begin{infobox}
\textbf{Security Note:}
Returns success message regardless of email existence to prevent email enumeration attacks.
\end{infobox}

\subsection{Get Current User}

\textbf{GET} \texttt{/api/v1/auth/me}

\subsubsection{Description}
Retrieve current authenticated user's profile information.

\subsubsection{Authentication}
Required: Bearer Token in Authorization header

\subsubsection{Use Case}
Fetch user profile data for dashboard display, settings, and personalization.

\subsection{Logout}

\textbf{POST} \texttt{/api/v1/auth/logout}

\subsubsection{Description}
Logout user and invalidate session token.

\subsubsection{Use Case}
Secure logout process that revokes the session token, preventing further API access with that token.

% Workflows Module
\chapter{Workflows Module}

\section{Module Overview}

The Workflows module manages automation workflows including creation, execution, versioning, and sharing capabilities.

\subsection{Base Path}
\texttt{/api/v1/workflows}

\subsection{Module Tag}
\texttt{Workflows}

\section{Endpoints}

\subsection{Create Workflow}

\textbf{POST} \texttt{/api/v1/workflows}

\subsubsection{Description}
Create a new automation workflow with nodes, edges, and variables.

\subsubsection{Authentication}
Required: Bearer Token

\subsubsection{Use Case}
Users create custom workflows by defining nodes (actions), edges (connections between actions), and variables (data to pass between nodes). Workflows can be private or publicly listed.

\subsubsection{Request Body}
\begin{lstlisting}[style=jsoncode,language=json]
{
  "name": "Email Campaign Workflow",
  "description": "Automated email campaign with user segmentation",
  "canBeListed": false,
  "nodes": [
    {
      "id": "node_1",
      "type": "trigger",
      "config": {...}
    },
    {
      "id": "node_2",
      "type": "action",
      "config": {...}
    }
  ],
  "edges": [
    {
      "source": "node_1",
      "target": "node_2"
    }
  ],
  "variables": {
    "emailSubject": "Welcome to NexAgent",
    "senderEmail": "noreply@example.com"
  }
}
\end{lstlisting}

\begin{infobox}
\textbf{Workflow Limits:}
\begin{itemize}[leftmargin=*]
    \item Name: 3-100 characters
    \item Maximum nodes: Depends on subscription plan
    \item Execution timeout: 5 minutes (configurable)
    \item Rate limit: 20 requests per minute
    \item HTTP Status: 201 Created on success
\end{itemize}
\end{infobox}

\subsection{Get Workflows}

\textbf{GET} \texttt{/api/v1/workflows}

\subsubsection{Description}
Retrieve paginated list of user's workflows with filtering options.

\subsubsection{Query Parameters}
\begin{itemize}[leftmargin=*]
    \item \texttt{page}: Page number (default: 1)
    \item \texttt{page\_size}: Items per page (1-100, default: 20)
    \item \texttt{status}: Filter by status (draft, active, archived)
\end{itemize}

\subsubsection{Use Case}
Dashboard display of user's workflows with pagination and filtering for efficient management of large workflow collections.

\subsection{Get Workflow by ID}

\textbf{GET} \texttt{/api/v1/workflows/\{workflow\_id\}}

\subsubsection{Description}
Retrieve detailed information about a specific workflow.

\subsubsection{Authorization}
User must own the workflow or workflow must be publicly listed.

\subsection{Update Workflow}

\textbf{PUT} \texttt{/api/v1/workflows/\{workflow\_id\}}

\subsubsection{Description}
Update workflow properties, nodes, edges, or variables. Increments version number.

\subsubsection{Use Case}
Modify existing workflows while maintaining version history for rollback capability.

\begin{infobox}
\textbf{Versioning:}
Each update increments the workflow version number, allowing users to track changes and potentially rollback to previous versions.
\end{infobox}

\subsection{Delete Workflow}

\textbf{DELETE} \texttt{/api/v1/workflows/\{workflow\_id\}}

\subsubsection{Description}
Permanently delete a workflow. Only workflow owner can delete.

\subsubsection{Use Case}
Remove unwanted or test workflows from account.

\subsection{Get Public Workflows}

\textbf{GET} \texttt{/api/v1/workflows/public/list}

\subsubsection{Description}
Browse publicly shared workflows from the community.

\subsubsection{Authentication}
Not required

\subsubsection{Use Case}
Discover and explore workflows shared by other users for inspiration or cloning.

Create a new notification with multi-channel delivery.

\subsubsection{Request Body}
\begin{lstlisting}[style=jsoncode,language=json]

  "user_id": "user_123",

\subsubsection{Use Case}
Display user's notification center with filtering and pagination.

\subsection{Mark as Read}

\textbf{POST} \texttt{/api/v1/notifications/mark-read}

\subsubsection{Description}
Mark one or multiple notifications as read.

\subsubsection{Use Case}
Batch update notification status when user reviews notification center.

\textbf{GET} \texttt{/api/v1/notifications/preferences}

\chapter{Glossary}

\begin{description}[leftmargin=!,labelwidth=\widthof{\bfseries Integration}]
    \item[Workflow] An automated sequence of actions and conditions
    \item[Template] Pre-built workflow available for cloning
    \item[Nexa] Marketplace item (workflow, plugin, or service)
    \item[Integration] Third-party service connection
    \item[Node] Individual action or trigger in a workflow
    \item[Edge] Connection between workflow nodes
    \item[Session] Authenticated user session with time-limited validity
    \item[Bearer Token] Authentication token passed in Authorization header
    \item[Rate Limit] Maximum requests allowed per time window
    \item[Webhook] HTTP callback for real-time event notifications
\end{description}
